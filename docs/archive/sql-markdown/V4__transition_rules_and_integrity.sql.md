-- D2 v0.4 / proposed Flyway V4; PostgreSQL 16+
-- Seed transition graph. Domain guards and event atomicity remain application responsibilities.
INSERT INTO state_transition_rules(entity_type,from_state,to_state,requires_guard)
SELECT entity_type, split_part(edge,'>',1), split_part(edge,'>',2), 'DOMAIN_GUARD'
FROM (VALUES
 ('TASK', 'BACKLOG>READY BACKLOG>BLOCKED BACKLOG>CANCELLED READY>IN_PROGRESS READY>BLOCKED READY>DEFERRED READY>CANCELLED BLOCKED>READY BLOCKED>DEFERRED BLOCKED>CANCELLED DEFERRED>READY DEFERRED>BLOCKED DEFERRED>CANCELLED IN_PROGRESS>VERIFYING IN_PROGRESS>BLOCKED IN_PROGRESS>FAILED IN_PROGRESS>CANCELLED VERIFYING>DONE VERIFYING>IN_PROGRESS VERIFYING>FAILED VERIFYING>CANCELLED FAILED>READY FAILED>CANCELLED'),
 ('OPERATION', 'READY>LEASED READY>WAITING_CAPACITY READY>BLOCKED_APPROVAL READY>CANCELLED WAITING_CAPACITY>READY WAITING_CAPACITY>CANCELLED BLOCKED_APPROVAL>READY BLOCKED_APPROVAL>CANCELLED LEASED>RUNNING LEASED>RECONCILING LEASED>CANCELLED RUNNING>VERIFYING RUNNING>RECONCILING RUNNING>FAILED RUNNING>CANCELLED VERIFYING>COMPLETED VERIFYING>RETRY_PENDING VERIFYING>FAILED VERIFYING>RECONCILING RETRY_PENDING>READY RETRY_PENDING>CANCELLED RECONCILING>VERIFYING RECONCILING>COMPLETED RECONCILING>RETRY_PENDING RECONCILING>FAILED RECONCILING>BLOCKED_APPROVAL FAILED>RETRY_PENDING'),
 ('INCIDENT', 'OPEN>TRIAGED TRIAGED>INVESTIGATING INVESTIGATING>FIXING INVESTIGATING>VERIFYING FIXING>VERIFYING FIXING>INVESTIGATING VERIFYING>RESOLVED VERIFYING>FIXING VERIFYING>INVESTIGATING RESOLVED>REOPENED REOPENED>INVESTIGATING'),
 ('CAPABILITY_VERSION', 'PLANNED>IMPLEMENTED IMPLEMENTED>VERIFIED IMPLEMENTED>INVALIDATED VERIFIED>AVAILABLE VERIFIED>INVALIDATED AVAILABLE>INVALIDATED INVALIDATED>IMPLEMENTED')
) AS rules(entity_type,edges)
CROSS JOIN LATERAL regexp_split_to_table(rules.edges,'\s+') AS edge
ON CONFLICT DO NOTHING;

ALTER TABLE operations ADD CONSTRAINT ck_operation_lease_generation_nonnegative CHECK(lease_generation >= 0);
ALTER TABLE execution_attempts ADD CONSTRAINT ck_attempt_lease_generation_positive CHECK(lease_generation > 0);
ALTER TABLE capacity_budgets ADD CONSTRAINT uq_budget_identity UNIQUE(id,planning_run_id,resource_id);
ALTER TABLE capacity_allocations ADD COLUMN budget_id uuid;
ALTER TABLE capacity_allocations ADD CONSTRAINT fk_allocation_budget_same_run_resource
 FOREIGN KEY(budget_id,planning_run_id,resource_id)
 REFERENCES capacity_budgets(id,planning_run_id,resource_id) NOT VALID;
CREATE INDEX idx_allocations_budget ON capacity_allocations(budget_id) WHERE budget_id IS NOT NULL;
CREATE INDEX idx_incidents_open_severity ON incidents(severity,detected_at) WHERE status NOT IN ('RESOLVED');
